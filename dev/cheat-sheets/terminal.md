# BONUP TERMINAL CHEAT SHEET

## WHERE AM I?
pwd

## LIST FILES
ls
ls -l
ls -la

## CHANGE DIRECTORY
cd foldername
cd ..
cd ~
cd ~/bonup-blackboard

## CREATE
mkdir foldername
mkdir -p path/to/folder
touch filename.py

## MOVE
mv oldname newname
mv file.txt folder/
mv folder1 folder2/

## COPY
cp file.txt backup.txt
cp -r folder1 folder2

## DELETE (DANGEROUS)
rm file.txt
rm -r folder
rm -rf folder  # nuclear

## CLEAR SCREEN
clear
